class ExemploCompleto {
    int a, b;
    float x;
    string nome;
    {
        a := 10;
        b := a + 5;
        x := a / 2.0;
        if (a > b) {
            b := a;
        } else {
            a := b;
        };
        do {
            a := a - 1;
        } while (a > 0);
        repeat {
            b := b * 2;
        } until (b >= 100);
        read(nome);
        write("Valor de a: " + a);
        write("Valor de b: " + b);
        write(x);
        write(nome);
    }
}
